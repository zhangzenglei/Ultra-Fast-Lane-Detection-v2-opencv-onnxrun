cmake_minimum_required(VERSION 3.16)
project(ufldv2_cpp_gpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(_ort_default "/home/zenglei/opt/onnx-gpu/onnxruntime-linux-x64-gpu-1.15.1")
if(DEFINED ENV{ORT_ROOT})
  set(_ort_default "$ENV{ORT_ROOT}")
endif()
set(ORT_ROOT "${_ort_default}" CACHE PATH "ONNX Runtime C++ GPU SDK root")
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
if(NOT EXISTS "${ORT_ROOT}/include/onnxruntime_cxx_api.h" OR
   NOT EXISTS "${ORT_ROOT}/lib/libonnxruntime.so")
  message(FATAL_ERROR "ORT SDK not found at ${ORT_ROOT}; set -DORT_ROOT=/path/to/sdk")
endif()
add_executable(ufldv2_gpu onnxruntime/main.cpp)
target_include_directories(ufldv2_gpu PRIVATE "${ORT_ROOT}/include" ${OpenCV_INCLUDE_DIRS} ${JSON_INCLUDE_DIR})
target_link_libraries(ufldv2_gpu PRIVATE "${ORT_ROOT}/lib/libonnxruntime.so" ${OpenCV_LIBS})
target_compile_options(ufldv2_gpu PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
set_target_properties(ufldv2_gpu PROPERTIES BUILD_RPATH "${ORT_ROOT}/lib")
